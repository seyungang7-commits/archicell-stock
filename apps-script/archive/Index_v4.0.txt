function store(key,value,session){
  const s=session?window.sessionStorage:window.localStorage;
  try{if(value===undefined){const raw=s.getItem(key);return raw?JSON.parse(raw):null;}
    if(value===null)s.removeItem(key);else s.setItem(key,JSON.stringify(value));return true;}catch(e){return value===undefined?null:false;}
}
function authSaved(value){

/* 조회용 호출: 제한 시간 안에 답이 없으면 실패로 처리합니다. (저장은 sendPending에서 따로 다룹니다) */
function call(fn,args,timeout){
  return new Promise((resolve,reject)=>{
    let done=false;const t=setTimeout(()=>{if(done)return;done=true;const e=new Error('응답이 늦어지고 있습니다. 연결 상태를 확인한 뒤 다시 시도해 주세요.');e.timeout=true;reject(e);},timeout||APP.timeoutMs);
    try{
      const run=google.script.run.withSuccessHandler(r=>{if(done)return;done=true;clearTimeout(t);resolve(r);})
        .withFailureHandler(e=>{if(done)return;done=true;clearTimeout(t);reject(e instanceof Error?e:new Error(String(e&&e.message||e)));});
      run[fn](...args);
    }catch(e){if(!done){done=true;clearTimeout(t);reject(e);}}
  });
}

  else rows=stockSortDefault(list).map(x=>({it:x.it,label:x.r.label}));
  return {active:active,counts:counts,rows:rows};
}

/* 저장 요청 보내기. 같은 요청 번호로 다시 보내도 서버가 한 번만 기록합니다(요청ID + 내용 서명 확인). */
function sendPending(){
  if(!pending||busy)return;
  if(pending.actor!==accessActor){msg('다른 작업자가 보낸 저장 요청이 남아 있습니다. 메뉴에서 원래 작업자로 바꾼 뒤 결과를 확인하세요.',true);return;}
  if(navigator.onLine===false){pendingSlow=false;toast('인터넷 연결을 확인하세요','err');controls();return;}
  busy=true;pendingSlow=false;const p=pending,rid=p.payload.requestId,t0=now();controls();
  clearTimeout(pendingTimer);
  pendingTimer=setTimeout(()=>{if(pending&&pending.payload.requestId===rid&&busy){busy=false;pendingSlow=true;toast('응답이 늦어지고 있습니다','err');controls();}},APP.timeoutMs);
  const mine=()=>pending&&pending.payload.requestId===rid;
  try{
    google.script.run.withSuccessHandler(packet=>{
      if(!mine())return;clearTimeout(pendingTimer);busy=false;pending=null;pendingSlow=false;session(null);
      METRIC.save={round:Math.round(now()-t0),server:packet&&packet.saveMs,at:Date.now()};
      onSaved(p,packet||{});
    }).withFailureHandler(e=>{
      if(!mine())return;clearTimeout(pendingTimer);busy=false;
      if(authFailure(e)){toastHide();controls();return;}
      if(isChecked(e)){ // 서버가 검증 후 거절: 기록되지 않았습니다. 입력값은 그대로 둡니다.
        pending=null;session(null);const t=cleanErr(e);
        if(sheetOpen)sheetErr=t;else msg(t,true);toast('저장되지 않았습니다','err');vibrate([80,60,80]);controls();load(true,true);
      }else{toast('저장 결과를 확인하지 못했습니다','err');controls();}
    }).uiMutate(p.method,p.payload,accessToken,{fast:APP.fastSave});
  }catch(e){clearTimeout(pendingTimer);busy=false;toast('저장 요청을 보내지 못했습니다','err');controls();}
}

/* 긴 쪽 1600px 이하 JPEG로 줄입니다(원본 비율 유지, 원본보다 키우지 않음). 서버 한도(약 2.5MB)를 넘으면 화질→크기 순으로 낮춥니다. */
function photoEncode(img){
  let k=Math.min(1,1600/Math.max(img.naturalWidth,img.naturalHeight)),q=.88,w=0,h=0,out='';
  for(let n=0;n<8;n++){
    const c=document.createElement('canvas');w=Math.max(1,Math.round(img.naturalWidth*k));h=Math.max(1,Math.round(img.naturalHeight*k));c.width=w;c.height=h;
    const x=c.getContext('2d');x.fillStyle='#fff';x.fillRect(0,0,w,h);x.drawImage(img,0,0,w,h);
    out=c.toDataURL('image/jpeg',q);
    if(out.length<3300000)return {url:out,w:w,h:h};
    if(q>0.7)q-=.09;else k*=.8;
  }
  throw Error('사진 용량이 너무 큽니다. 더 작은 사진으로 골라 주세요.');
}
